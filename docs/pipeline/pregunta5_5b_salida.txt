function detectAutoTrigger(r){
  if (!r || r.error || r.hardNo) return null;
  if (r.globalVerdict==='NO' || r.globalVerdict==='N/A') return null;
  const applicable = r.strategies.filter(s=>s.applicable);

  const byGroup = {};
  for (const s of applicable){
    if (s.score>=80) (byGroup[s.group] = byGroup[s.group]||[]).push(s);
  }
  for (const group in byGroup){
    if (byGroup[group].length>=2) return {type:'AUTO_MULTI', strategies: byGroup[group]};
  }

  const highSingle = applicable.filter(s=>s.score>=90);
  if (highSingle.length){
    const top = highSingle.reduce((a,b)=> b.score>a.score ? b : a, highSingle[0]);
    return {type:'AUTO_HIGH', strategies:[top]};
  }
  return null;
}

async function autoCaptureSnapshots(evaluated){
  let captured = 0;
  for (const r of evaluated){
    if (r.error) continue;
    const trigger = detectAutoTrigger(r);
    if (!trigger) continue;
    if (state.followedToday.has(r.ticker)) continue;
    const payload = buildSnapshotPayload(r, trigger.type, trigger.strategies);
    if (!payload) continue;
    const result = await postSnapshot(payload);
    if (result && result.created){
      captured++;
      state.followedToday.add(r.ticker);
    }
  }
  return captured;
}

function evaluateTicker(ticker, ind, funda, mode, settings, btcGateOn, insiderSummary){

  if (!d) return null;
  const hardNo = tickerHardNo(mode, d.price, d, settings);

  const fns = mode==='NYSE' ? NYSE_STRATEGIES : CRYPTO_STRATEGIES;

function pctChangeSeries(candles, lookback){
  lookback = lookback||60;
  const closes = candles.map(c=>c.c).slice(-(lookback+1));
  const out = [];
  for (let i=1;i<closes.length;i++) out.push((closes[i]-closes[i-1])/closes[i-1]);
  return out;
}

function pearsonCorrelation(a, b){
  const n = Math.min(a.length, b.length);
  if (n < 10) return NaN;
  const av = a.slice(-n), bv = b.slice(-n);
  const meanA = av.reduce((x,y)=>x+y,0)/n, meanB = bv.reduce((x,y)=>x+y,0)/n;
  let num=0, denA=0, denB=0;
  for (let i=0;i<n;i++){
    const da = av[i]-meanA, db = bv[i]-meanB;
    num += da*db; denA += da*da; denB += db*db;
  }
  const den = Math.sqrt(denA*denB);
  return den>0 ? num/den : NaN;
}

// A1 — matriz de correlación entre señales activas (veredicto != NO/N-A) del scan actual.
function computeCorrelationMatrix(results){
  const active = results.filter(r=>!r.error && r.globalVerdict!=='NO' && r.globalVerdict!=='N/A'
    && r.ind['1d'] && r.ind['1d'].candles && r.ind['1d'].candles.length>15);
  const series = {};
  for (const r of active) series[r.ticker] = pctChangeSeries(r.ind['1d'].candles, 60);
  const tickers = Object.keys(series);
  const matrix = {};
  for (const t of tickers) matrix[t] = {};
  for (let i=0;i<tickers.length;i++){
    for (let j=i+1;j<tickers.length;j++){
      const c = pearsonCorrelation(series[tickers[i]], series[tickers[j]]);
      matrix[tickers[i]][tickers[j]] = c;
      matrix[tickers[j]][tickers[i]] = c;
    }
  }
  return matrix;
}

function getCorrelatedPeers(matrix, ticker, threshold){
  threshold = threshold==null ? 0.75 : threshold;
  const row = matrix[ticker];
  if (!row) return [];
  return Object.entries(row)
    .filter(([t,c])=> c!=null && !isNaN(c) && Math.abs(c) > threshold)
    .map(([t,c])=>({ticker:t, corr:c}))
    .sort((a,b)=>Math.abs(b.corr)-Math.abs(a.corr));
}

// Marca el aviso SOLO en la señal de menor score del cluster (regla operativa:
// toma solo la de mayor score, o divide el sizing entre las correlacionadas).
function annotateCorrelationWarnings(results, matrix){
  const scoreByTicker = {};
  for (const r of results) if (!r.error) scoreByTicker[r.ticker] = r.globalScore;
  for (const r of results){
    if (r.error){ r.correlatedPeers = []; r.correlationWarning = null; continue; }
    const peers = getCorrelatedPeers(matrix, r.ticker);
    r.correlatedPeers = peers;
    const higherPeer = peers.find(p => (scoreByTicker[p.ticker]||0) > r.globalScore);
    r.correlationWarning = higherPeer ? {peerTicker: higherPeer.ticker, corr: higherPeer.corr} : null;
  }
}

async def scan_batch(req: ScanBatchRequest):
    if len(req.tickers) > MAX_BATCH_TICKERS:
        raise HTTPException(
            status_code=400,
            detail=f"máximo {MAX_BATCH_TICKERS} tickers por batch",
        )
    tfs = _validate_timeframes(req.timeframes)

    semaphore = asyncio.Semaphore(BATCH_WORKERS)

    async def _process(raw_ticker: str) -> dict:
        async with semaphore:
            try:
                ticker = _validate_ticker(raw_ticker)
                data = await get_ticker_data(ticker, tfs)
                return {
                    "ticker": ticker,
                    "timestamp": datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),

            except Exception as e:
                logger.warning(f"batch error for {raw_ticker}: {e}")

function applyEventAdjustments(strategies, mode, funda, insiderSummary, hardNo, rvol1d, rvolMin){
  let earningsDays = null;
  if (mode==='NYSE' && funda && funda.next_earnings){
    const ed = new Date(funda.next_earnings+'T00:00:00Z');
    earningsDays = (ed - Date.now()) / 86400000;
  }
  const insiderScore = (mode==='NYSE' && insiderSummary)
    ? Math.max(-8, Math.min(8, insiderSummary.insider_score||0)) : 0;

  return strategies.map(s=>{
    if (!s.applicable) return s;
    // s.score ya está normalizado a base 100 (mkResult) — los modificadores
    // de evento se aplican DESPUÉS de normalizar, nunca dentro del raw.
    let score = s.score;
    const factors = s.factors.slice();
    let earningsAdj = 0, insiderAdj = 0;

    if (earningsDays!=null && earningsDays>=0 && (s.group==='swing' || s.group==='medio')){
      const maxHold = STRATEGY_META[s.id].maxHoldDays;
      if (earningsDays <= maxHold){
        earningsAdj = -10;
        score += earningsAdj;
        factors.push(f(`Earnings en ${Math.ceil(earningsDays)}d dentro del holding — cierra antes`, 'EVENTO', -10));
      }
    }
    if (insiderScore !== 0){
      insiderAdj = insiderScore;
      score += insiderAdj;
      factors.push(f(`Insider score ${insiderScore>=0?'+':''}${insiderScore}`, 'INSIDER', insiderScore));
    }
    if (score === s.score) return s;

    score = Math.max(0, Math.min(100, score));
    const verdict = finalizeVerdict(score, hardNo, rvol1d, rvolMin);
    // s.baseScore se conserva tal cual (score normalizado antes de modificadores)
    // para poder mostrar el desglose "78 (base 82 · insider +4 · earnings -8)".
    return Object.assign({}, s, {score, earningsAdj, insiderAdj, factors, verdict});
  });
}
